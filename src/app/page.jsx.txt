'use client';
import { useState, useEffect } from 'react';
import { AlertTriangle, Clock, PlusCircle, Filter } from 'lucide-react';

export default function Dashboard() {
  const [devices, setDevices] = useState([]);
  const [filterType, setFilterType] = useState('all');
  const [decisionModal, setDecisionModal] = useState(false);
  const [decisionResult, setDecisionResult] = useState('Kullanıma Uygun');
  const [explanation, setExplanation] = useState('');
  const [errorMsg, setErrorMsg] = useState('');

  useEffect(() => {
    fetch(`/api/devices?filter=${filterType}`)
      .then((res) => res.json())
      .then((data) => setDevices(Array.isArray(data) ? data : []))
      .catch((err) => console.error('Veri çekilemedi:', err));
  }, [filterType]);

  const handleDecisionSubmit = (e) => {
    e.preventDefault();
    if (decisionResult === 'Sınırlı Kullanım' && !explanation.trim()) {
      setErrorMsg('Sınırlı Kullanım seçildiğinde açıklama girmek ZORUNLUDUR!');
      return;
    }
    alert('Kalibrasyon kararı başarıyla kaydedildi!');
    setDecisionModal(false);
    setExplanation('');
    setErrorMsg('');
  };

  return (
    <div className="p-4 sm:p-6 max-w-7xl mx-auto space-y-6">
      
      {/* Özet Kartlar */}
      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
        <div 
          onClick={() => setFilterType('all')}
          className="bg-white p-4 rounded-xl shadow-sm border border-slate-200 cursor-pointer active:scale-95 transition"
        >
          <div className="flex justify-between items-center">
            <span className="text-slate-500 text-sm font-medium">Toplam Ekipman</span>
            <Filter className="text-blue-500" size={18} />
          </div>
          <h2 className="text-2xl font-bold mt-1 text-slate-900">{devices.length}</h2>
        </div>

        <div 
          onClick={() => setFilterType('upcoming')}
          className="bg-white p-4 rounded-xl shadow-sm border border-slate-200 cursor-pointer active:scale-95 transition"
        >
          <div className="flex justify-between items-center">
            <span className="text-slate-500 text-sm font-medium">Yaklaşanlar (30 Gün)</span>
            <Clock className="text-amber-500" size={18} />
          </div>
          <h2 className="text-2xl font-bold mt-1 text-amber-600">Filtrele</h2>
        </div>

        <div 
          onClick={() => setFilterType('expired')}
          className="bg-white p-4 rounded-xl shadow-sm border border-slate-200 cursor-pointer active:scale-95 transition"
        >
          <div className="flex justify-between items-center">
            <span className="text-slate-500 text-sm font-medium">Süresi Geçenler</span>
            <AlertTriangle className="text-red-500" size={18} />
          </div>
          <h2 className="text-2xl font-bold mt-1 text-red-600">Kritik</h2>
        </div>
      </div>

      {/* Tablo Alanı */}
      <div className="bg-white rounded-xl shadow-sm border border-slate-200 overflow-hidden">
        <div className="p-4 sm:p-5 border-b border-slate-100 flex flex-col sm:flex-row justify-between items-start sm:items-center gap-3">
          <h3 className="font-semibold text-base sm:text-lg text-slate-800">Cihaz Listesi ({filterType.toUpperCase()})</h3>
          <button 
            onClick={() => setDecisionModal(true)}
            className="w-full sm:w-auto bg-blue-600 hover:bg-blue-700 text-white px-4 py-2 rounded-lg text-sm font-medium flex items-center justify-center gap-2 transition"
          >
            <PlusCircle size={16} /> Kalibrasyon Kararı Gir
          </button>
        </div>

        <div className="overflow-x-auto">
          <table className="w-full text-left border-collapse min-w-[600px]">
            <thead>
              <tr className="bg-slate-50 text-slate-600 text-xs uppercase tracking-wider border-b border-slate-200">
                <th className="p-3 sm:p-4">Kod</th>
                <th className="p-3 sm:p-4">Cihaz Adı</th>
                <th className="p-3 sm:p-4">İlk Alınma</th>
                <th className="p-3 sm:p-4">Gelecek Kalibrasyon</th>
                <th className="p-3 sm:p-4">Durum</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-slate-100 text-xs sm:text-sm">
              {devices.length === 0 ? (
                <tr>
                  <td colSpan="5" className="p-6 text-center text-slate-400">Kayıt bulunamadı.</td>
                </tr>
              ) : (
                devices.map((dev) => (
                  <tr key={dev.id} className="hover:bg-slate-50">
                    <td className="p-3 sm:p-4 font-medium text-slate-900">{dev.device_code}</td>
                    <td className="p-3 sm:p-4 text-slate-700">{dev.device_name}</td>
                    <td className="p-3 sm:p-4 text-slate-600">{dev.first_purchase_date}</td>
                    <td className="p-3 sm:p-4 font-semibold text-blue-600">{dev.next_calibration_date}</td>
                    <td className="p-3 sm:p-4">
                      <span className="bg-emerald-50 text-emerald-700 px-2 py-1 rounded-full text-xs font-medium">Aktif</span>
                    </td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>

      {/* Modal Ekranı */}
      {decisionModal && (
        <div className="fixed inset-0 bg-slate-900/50 backdrop-blur-sm flex items-center justify-center p-4 z-50">
          <div className="bg-white rounded-xl shadow-xl max-w-md w-full p-6 space-y-4">
            <h3 className="text-lg font-bold text-slate-900">Kalibrasyon Kararı</h3>
            
            {errorMsg && (
              <div className="bg-red-50 border border-red-200 text-red-600 p-3 rounded-lg text-xs font-medium">
                {errorMsg}
              </div>
            )}

            <form onSubmit={handleDecisionSubmit} className="space-y-4">
              <div>
                <label className="block text-xs font-medium text-slate-700 mb-1">Karar Seçiniz</label>
                <select 
                  value={decisionResult} 
                  onChange={(e) => setDecisionResult(e.target.value)}
                  className="w-full border border-slate-300 rounded-lg p-2.5 text-sm bg-white"
                >
                  <option value="Kullanıma Uygun">Kullanıma Uygun</option>
                  <option value="Sınırlı Kullanım">Sınırlı Kullanım (Açıklama Zorunlu)</option>
                  <option value="Kullanım Dışı">Kullanım Dışı</option>
                </select>
              </div>

              <div>
                <label className="block text-xs font-medium text-slate-700 mb-1">
                  Açıklama {decisionResult === 'Sınırlı Kullanım' && <span className="text-red-500">*Zorunlu</span>}
                </label>
                <textarea 
                  rows="3"
                  value={explanation}
                  onChange={(e) => setExplanation(e.target.value)}
                  placeholder="Gerekçeyi yazınız..."
                  className="w-full border border-slate-300 rounded-lg p-2 text-sm"
                ></textarea>
              </div>

              <div className="flex justify-end gap-3 pt-2">
                <button 
                  type="button" 
                  onClick={() => setDecisionModal(false)}
                  className="px-4 py-2 border border-slate-300 text-slate-700 rounded-lg text-sm font-medium"
                >
                  İptal
                </button>
                <button 
                  type="submit" 
                  className="px-4 py-2 bg-blue-600 text-white rounded-lg text-sm font-medium hover:bg-blue-700"
                >
                  Kaydet
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

    </div>
  );
}
